# Calculo de area do triangulo
base = float(input("Digite o valor da base: "))
altura = float(input("Digite o valor da altura: "))
area = (base * altura) / 2

# Formatacao de numeros ':.{qtdCasas}.{tipo}
print(f"O valor da área é {area:.2f}")
